Port this Python function to idiomatic Go, returning a time.Duration and an error, without time.ParseDuration. First list the inputs on which the Python version behaves questionably — "1.2.3s", non-ASCII digits that isdigit accepts, an empty string — and say what the Go version should do instead. Then write the port, table-driven tests for every case, and a fuzz test that it never panics. Below are the whole module the function lives in, the test file that passes against it today, the config loader, the config file both callers read, the scheduler loop, the database half of the scrub, the unit file as it was deployed on the night of the incident, the journal of that night and the morning after, the REPL session in which the questionable inputs were probed on the box, the Go file the port will live beside with its test, the CLI entry point, and the packaging.

The module, retention/duration.py, in full:

~~~python
"""Durations the way the config files write them: "1h30m", "45s", "2.5m".

One format, parsed by everything in this package and formatted back for
logs. The unit set is deliberately small — hours, minutes, seconds,
milliseconds — because that is all the schedules use.
"""

# For callers that need the unit set without parsing (config validation).
UNIT_SECONDS = {
    "h": 3600.0,
    "m": 60.0,
    "s": 1.0,
    "ms": 0.001,
}

__all__ = ["format_duration", "parse_duration"]


def parse_duration(s: str) -> float:
    """Parse "1h30m", "45s", "2.5m" or "500ms" into seconds."""
    units = {"h": 3600, "m": 60, "s": 1, "ms": 0.001}
    total, num, i = 0.0, "", 0
    while i < len(s):
        ch = s[i]
        if ch.isdigit() or ch == ".":
            num += ch
            i += 1
            continue
        unit = s[i:i+2] if s[i:i+2] == "ms" else ch
        if unit not in units or not num:
            raise ValueError(f"bad duration: {s!r}")
        total += float(num) * units[unit]
        num = ""
        i += len(unit)
    if num:
        raise ValueError(f"missing unit in {s!r}")
    return total


def format_duration(seconds: float) -> str:
    """Format seconds the way the config files write them."""
    parts = []
    for unit, size in (("h", 3600), ("m", 60)):
        whole, seconds = divmod(seconds, size)
        if whole:
            parts.append(f"{int(whole)}{unit}")
    if seconds.is_integer():
        if seconds or not parts:
            parts.append(f"{int(seconds)}s")
    else:
        parts.append(f"{seconds:g}s")
    return "".join(parts)
~~~

The config loader, retention/config.py:

~~~python
"""Load retention.toml: three durations and nothing else."""

import tomllib
from pathlib import Path

from .duration import parse_duration

REQUIRED_KEYS = ("interval", "retention", "grace_period")


class ConfigError(Exception):
    """A retention file is missing a key or carries a bad duration."""


def load_config(path):
    """Return {key: seconds} for the three required durations."""
    path = Path(path)
    with path.open("rb") as f:
        raw = tomllib.load(f)
    missing = [k for k in REQUIRED_KEYS if k not in raw]
    if missing:
        raise ConfigError(f"{path}: missing keys: {', '.join(missing)}")
    out = {}
    for key in REQUIRED_KEYS:
        try:
            out[key] = parse_duration(raw[key])
        except ValueError as e:
            raise ConfigError(f"{path}: key {key!r}: {e}") from e
    return out
~~~

The sweep loop, retention/scheduler.py — the environment handling in load_interval is where September's incident lives:

~~~python
"""The scrub loop: sleep the configured interval, delete past retention."""

import logging
import os
import time
from pathlib import Path

from .config import load_config
from .duration import parse_duration

log = logging.getLogger("retention.scheduler")

DEFAULT_CONFIG = Path("/etc/retention/retention.toml")


def load_interval(config_path=DEFAULT_CONFIG):
    """The sweep interval: $SCRUB_INTERVAL if it is set, the file otherwise.

    "If it is set" is presence, not truth: an empty value is a value, so
    an operator can distinguish SCRUB_INTERVAL="" from no override at
    all. That distinction is why the environment is read this way.
    """
    override = os.environ.get("SCRUB_INTERVAL")
    if override is not None:
        return parse_duration(override)
    return parse_duration(load_config(config_path)["interval"])


def run(config_path=DEFAULT_CONFIG):
    config = load_config(config_path)
    interval = load_interval(config_path)
    log.info(
        "starting: interval %.3fs, retention %.0fs",
        interval, config["retention"],
    )
    passes = 0
    while True:
        started = time.monotonic()
        scanned, deleted = scrub(config)
        passes += 1
        took = time.monotonic() - started
        log.info(
            "scrub pass %d in %.3fs: scanned %d rows past retention %.1fs, "
            "deleted %d; sleeping %.3fs",
            passes, took, scanned, config["retention"], deleted, interval,
        )
        time.sleep(interval)


def scrub(config):
    """One pass: count what is past retention, delete what is past it
    and the grace period too."""
    from . import warehouse  # imported late: it opens the database

    cutoff = time.time() - config["retention"]
    scanned = warehouse.count_events_before(cutoff)
    deleted = warehouse.delete_events_before(cutoff - config["grace_period"])
    return scanned, deleted
~~~

The database half, retention/warehouse.py:

~~~python
"""The database side of the scrub: count and delete old events."""

import os

import psycopg

_DSN = os.environ.get("WAREHOUSE_DSN", "dbname=warehouse")


def count_events_before(ts):
    with psycopg.connect(_DSN) as conn, conn.cursor() as cur:
        cur.execute("SELECT count(*) FROM events WHERE created_at < %s", (ts,))
        return cur.fetchone()[0]


def delete_events_before(ts):
    with psycopg.connect(_DSN) as conn, conn.cursor() as cur:
        cur.execute("DELETE FROM events WHERE created_at < %s", (ts,))
        return cur.rowcount
~~~

The Go file the port will live beside, internal/retention/retention.go — the idioms are the house style the port should keep:

~~~go
// Package retention is the Go side of the warehouse scrub: the policy
// a config file resolves to, and the errors a bad one produces. The
// duration parser joins it from the Python module of the same name.
package retention

import (
	"errors"
	"fmt"
	"strconv"
	"strings"
)

// ErrPolicy is the sentinel every error in this package wraps, so a
// caller can tell a bad policy file from a bad database.
var ErrPolicy = errors.New("retention: bad policy")

// ParseDays parses a plain whole-day count, the one unit the Python
// parser refused to grow. Negative counts and anything with a unit
// character in it are policy errors.
func ParseDays(s string) (int, error) {
	n, err := strconv.Atoi(strings.TrimSpace(s))
	if err != nil || n < 0 {
		return 0, fmt.Errorf("%w: day count %q", ErrPolicy, s)
	}
	return n, nil
}
~~~

Its test, internal/retention/retention_test.go:

~~~go
package retention

import (
	"errors"
	"testing"
)

func TestParseDays(t *testing.T) {
	tests := []struct {
		in      string
		want    int
		wantErr bool
	}{
		{in: "0", want: 0},
		{in: "14", want: 14},
		{in: " 7 ", want: 7},
		{in: "-1", wantErr: true},
		{in: "14d", wantErr: true},
		{in: "", wantErr: true},
		{in: "1e3", wantErr: true},
	}
	for _, tt := range tests {
		got, err := ParseDays(tt.in)
		if tt.wantErr {
			if !errors.Is(err, ErrPolicy) {
				t.Errorf("ParseDays(%q) error = %v, want ErrPolicy", tt.in, err)
			}
			continue
		}
		if err != nil || got != tt.want {
			t.Errorf("ParseDays(%q) = %d, %v, want %d, nil", tt.in, got, err, tt.want)
		}
	}
}
~~~

The CLI entry point, retention/__main__.py:

~~~python
"""`python3 -m retention` runs the scrub loop; given a config path, it
shows what that policy file resolves to instead."""

import argparse
import logging
import sys

from .config import ConfigError, load_config
from .duration import format_duration, parse_duration
from .scheduler import run

# Bare messages: journald already stamps every line with a time and a
# unit, so a level or logger prefix would only double them.
logging.basicConfig(level=logging.INFO, format="%(message)s")


def main(argv=None):
    parser = argparse.ArgumentParser(
        prog="retention",
        description="Print the policy a retention.toml resolves to.",
    )
    parser.add_argument("config", help="path to a retention.toml")
    parser.add_argument(
        "--interval",
        help='override the sweep interval, e.g. "45s" or "1h30m"',
    )
    args = parser.parse_args(argv)
    try:
        policy = load_config(args.config)
        if args.interval is not None:
            policy["interval"] = parse_duration(args.interval)
    except (ConfigError, ValueError) as e:
        return sys.exit(f"retention: {e}")
    for key in ("interval", "retention", "grace_period"):
        print(f"{key:14} {policy[key]:>10.3f}  ({format_duration(policy[key])})")
    return 0


if __name__ == "__main__":
    if len(sys.argv) == 1:
        run()  # the service form: sweep until the unit stops it
    else:
        sys.exit(main())
~~~
