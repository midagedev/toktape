/srv/orders-api/routes/parts.js, the read side the terminal browses:

~~~js
"use strict";

const express = require("express");
const { pool } = require("../db");

const router = express.Router();

// The terminal's catalogue list: one page at a time, recently touched
// parts first. The terminal polls page 1 while idle.
router.get("/parts", async (req, res, next) => {
  const page = Math.max(1, Number(req.query.page) || 1);
  const size = 50;
  try {
    const total = await pool.query("select count(*)::int as n from parts");
    const rows = await pool.query(
      `select sku, title, category, stock, unit_price_cents, updated_at
         from parts
        order by updated_at desc
        limit $1 offset $2`,
      [size, (page - 1) * size],
    );
    res.json({ total: total.rows[0].n, rows: rows.rows });
  } catch (err) {
    next(err);
  }
});

router.get("/parts/:sku", async (req, res, next) => {
  try {
    const found = await pool.query(
      `select sku, title, category, stock, unit_price_cents, updated_at
         from parts
        where sku = $1`,
      [req.params.sku],
    );
    if (found.rowCount === 0) {
      return res.status(404).json({ error: `no part with sku ${req.params.sku}` });
    }
    res.json(found.rows[0]);
  } catch (err) {
    next(err);
  }
});

module.exports = router;
~~~

/srv/orders-api/routes/orders.js, the write side that fails:

~~~js
"use strict";

const express = require("express");
const { logger } = require("../middleware/requestLog");
const { pool } = require("../db");

const router = express.Router();

// POST /v1/orders — one order, one part, a quantity. Small on purpose:
// the warehouse terminal posts one line at a time as the picker works.
router.post("/orders", async (req, res, next) => {
  const { sku, quantity } = req.body || {};
  if (typeof sku !== "string" || sku.length === 0) {
    return res.status(400).json({ error: "sku is required" });
  }
  if (!Number.isInteger(quantity) || quantity < 1 || quantity > 999) {
    return res.status(400).json({ error: "quantity must be an integer between 1 and 999" });
  }

  const client = await pool.connect();
  try {
    // One transaction per order: the row lock only exists between a
    // BEGIN and a COMMIT, so it has to span the select for update and
    // the stock update that must not race another picker's order.
    await client.query("begin");
    const part = await client.query(
      "select id, stock, unit_price_cents from parts where sku = $1 for update",
      [sku],
    );
    if (part.rowCount === 0) {
      await client.query("rollback");
      return res.status(404).json({ error: `no part with sku ${sku}` });
    }
    if (part.rows[0].stock < quantity) {
      await client.query("rollback");
      return res.status(409).json({ error: `only ${part.rows[0].stock} left of ${sku}` });
    }
    const inserted = await client.query(
      `insert into orders (part_id, quantity, unit_price_cents)
       values ($1, $2, $3) returning id`,
      [part.rows[0].id, quantity, part.rows[0].unit_price_cents],
    );
    await client.query("update parts set stock = stock - $1, updated_at = now() where id = $2", [
      quantity,
      part.rows[0].id,
    ]);
    await client.query("commit");
    logger.info({ orderId: inserted.rows[0].id, sku, quantity }, "order taken");
    return res.status(201).json({ id: inserted.rows[0].id, done: true });
  } catch (err) {
    await client.query("rollback").catch(() => {});
    return next(err);
  } finally {
    client.release();
  }
});

router.use((err, _req, res, _next) => {
  logger.error({ err }, "unhandled error");
  res.status(500).json({ error: "internal error" });
});

module.exports = router;
~~~

~~~js
"use strict";

const pino = require("pino");

const logger = pino({
  level: process.env.LOG_LEVEL || "info",
  base: null,
  timestamp: pino.stdTimeFunctions.isoTime,
});

// One line per finished response. Requests whose connection dies before
// the response is written never reach "finish", so they leave no line.
function requestLog(req, res, next) {
  const start = process.hrtime.bigint();
  res.on("finish", () => {
    logger.info(
      {
        method: req.method,
        path: req.originalUrl,
        status: res.statusCode,
        durationMs: Number(process.hrtime.bigint() - start) / 1e6,
      },
      "request",
    );
  });
  next();
}

module.exports = { requestLog, logger };
~~~
