~~~ts
// Shared shapes for the parts catalogue. The backend owns the canonical
// definitions; these mirror the JSON its endpoints return.

export type Category = "motor" | "controller" | "sensor" | "cable";

export interface Result {
  id: string;
  sku: string;
  title: string;
  category: Category;
  stock: number;
  unitPriceCents: number;
  updatedAt: string; // ISO 8601, from the server clock

// /api/parts/filter rows carry two fields the search results do not.
export interface PartRow extends Result {
  bin: string;
  reorderAt: number;
}

export interface FilterResponse {

  inStockOnly: boolean;
  total: number;
  rows: PartRow[];
}

export type RequestState = "idle" | "loading" | "error";
~~~

src/api/client.ts, the wrapper the rest of the app fetches through:

~~~ts
// Thin wrapper so every caller gets the same error shape. Nothing here
// adds a timeout or retries; callers own that decision.

export class ApiError extends Error {
  constructor(
    readonly status: number,
    readonly body: string,
  ) {
    super(`request failed with status ${status}`);
    this.name = "ApiError";
  }
}

export async function getJson<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(path, init);
  if (!res.ok) {
    throw new ApiError(res.status, await res.text());
  }
  return (await res.json()) as T;
}
~~~

src/hooks/useDebouncedValue.ts, a hook that already exists and is used elsewhere:

~~~ts
import { useEffect, useState } from "react";

// Returns `value` unchanged until it has been stable for `delayMs`. The
// timer is cleared on every change and on unmount, so a fast typist pays
// for one timer, not one per keystroke. FilterPanel is the current user.
export function useDebouncedValue<T>(value: T, delayMs: number): T {
  const [debounced, setDebounced] = useState(value);

  useEffect(() => {
    const id = setTimeout(() => setDebounced(value), delayMs);
    return () => clearTimeout(id);
  }, [value, delayMs]);

  return debounced;
}
~~~

src/components/FilterPanel.tsx, the panel next to the search box, which the team treated as the reference implementation:

~~~tsx
import { useEffect, useRef, useState } from "react";
import { ApiError, getJson } from "../api/client";
import type { Category, FilterResponse, PartRow, RequestState } from "../types";
import { useDebouncedValue } from "../hooks/useDebouncedValue";

const CATEGORIES: Category[] = ["motor", "controller", "sensor", "cable"];

export function FilterPanel({ onRows }: { onRows: (rows: PartRow[], total: number) => void }) {
  const [categories, setCategories] = useState<Category[]>([]);
  const [inStockOnly, setInStockOnly] = useState(false);
  const [state, setState] = useState<RequestState>("idle");
  const [error, setError] = useState<string | null>(null);

  const debouncedCategories = useDebouncedValue(categories, 300);
  const firstRender = useRef(true);

  useEffect(() => {
    // The page loads the unfiltered table itself; this panel only reacts
    // to the filters actually changing.
    if (firstRender.current) {
      firstRender.current = false;
      return;
    }

    const controller = new AbortController();

    async function load() {
      setState("loading");
      setError(null);
      try {
        const params = new URLSearchParams({ inStock: String(inStockOnly) });
        for (const c of debouncedCategories) {
          params.append("category", c);
        }
        const res = await getJson<FilterResponse>(`/api/parts/filter?${params}`, {
          signal: controller.signal,
        });
        setState("idle");
        onRows(res.rows, res.total);
      } catch (err) {
        if (err instanceof DOMException && err.name === "AbortError") {
          return; // a newer set of filters has taken over
        }
        setState("error");
        setError(
          err instanceof ApiError ? `filter request failed (${err.status})` : "filter request failed",
        );
      }
    }

    load();
    return () => controller.abort();
  }, [debouncedCategories, inStockOnly, onRows]);

  return (
    <fieldset className="filter">
      <legend>Filter</legend>
      {CATEGORIES.map(c => (
        <label key={c}>
          <input
            type="checkbox"
            checked={categories.includes(c)}
            onChange={e =>
              setCategories(prev => (e.target.checked ? [...prev, c] : prev.filter(x => x !== c)))
            }
          />
          {c}
        </label>
      ))}
      <label>
        <input type="checkbox" checked={inStockOnly} onChange={e => setInStockOnly(e.target.checked)} />
        in stock only
      </label>
      {state === "loading" && <p className="filter__status">filtering…</p>}
      {state === "error" && <p className="filter__status filter__status--error">{error}</p>}
    </fieldset>
  );
}
~~~

src/pages/PartsDirectory.tsx, the page that mounts the search box:

~~~tsx
import { useCallback, useEffect, useState } from "react";
import { ApiError, getJson } from "../api/client";
import type { FilterResponse, PartRow, RequestState } from "../types";
import { SearchBox } from "../components/SearchBox";
import { FilterPanel } from "../components/FilterPanel";
import { ErrorBanner } from "../components/ErrorBanner";

export function PartsDirectory() {
  const [rows, setRows] = useState<PartRow[]>([]);
  const [total, setTotal] = useState(0);
  const [state, setState] = useState<RequestState>("idle");
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    let cancelled = false;

    async function load() {
      try {
        const res = await getJson<FilterResponse>("/api/parts/filter");
        if (cancelled) return;
        setState("idle");
        setRows(res.rows);
        setTotal(res.total);
      } catch (err) {
        if (cancelled) return;
        setState("error");
        setError(
          err instanceof ApiError
            ? `could not load the catalogue (${err.status})`
            : "could not load the catalogue",
        );
      }
    }

    load();
    return () => {
      cancelled = true;
    };
  }, []);

  const handleFiltered = useCallback((next: PartRow[], nextTotal: number) => {
    setRows(next);
    setTotal(nextTotal);
  }, []);

  return (
    <main className="directory">
      <h1>Parts directory</h1>
      <SearchBox />
      <FilterPanel onRows={handleFiltered} />
      {state === "error" ? (
        <ErrorBanner message={error ?? "could not load the catalogue"} />
      ) : (
        <>
          <p className="directory__count">
            {rows.length} of {total} parts
          </p>
          <table className="directory__table">
            <thead>
              <tr>
                <th>SKU</th>
                <th>Title</th>
                <th>Category</th>
                <th>Stock</th>
                <th>Bin</th>
              </tr>
            </thead>
            <tbody>
              {rows.map(r => (
                <tr key={r.id}>
                  <td>{r.sku}</td>
                  <td>{r.title}</td>
                  <td>{r.category}</td>
                  <td className={r.stock <= r.reorderAt ? "directory__low" : undefined}>{r.stock}</td>
                  <td>{r.bin}</td>
                </tr>
              ))}
            </tbody>
          </table>
        </>
      )}
    </main>
  );
}
~~~

src/components/SearchBox.tsx, the component the ticket is about, in full:

~~~tsx
import { useEffect, useState } from "react";
import type { Result } from "../types";

export function SearchBox() {
  const [query, setQuery] = useState("");
  const [results, setResults] = useState<Result[]>([]);

  useEffect(() => {
    const id = setTimeout(async () => {
      const res = await fetch("/api/search?q=" + query);
      setResults(await res.json());
    }, 300);
  }, [query]);

  return (
    <div className="search-box">
      <input value={query} onChange={e => setQuery(e.target.value)} />
      <ul>{results.map(r => <li>{r.title}</li>)}</ul>
    </div>
  );
}
~~~

src/components/SearchBox.test.tsx, the suite that is green on main today:

~~~tsx
import { afterEach, beforeEach, describe, expect, it, vi } from "vitest";
import { act, fireEvent, render, screen } from "@testing-library/react";
import { SearchBox } from "./SearchBox";

const fetchMock = vi.fn();

function jsonResponse(results: { title: string }[]) {
  return { ok: true, json: () => Promise.resolve(results) };
}

function type(value: string) {
  fireEvent.change(screen.getByRole("textbox"), { target: { value } });
}

describe("SearchBox", () => {
  beforeEach(() => {
    vi.useFakeTimers();
    vi.stubGlobal("fetch", fetchMock);
    fetchMock.mockReset();
  });

  afterEach(() => {
    vi.useRealTimers();
    vi.unstubAllGlobals();
  });

  it("waits for a pause in typing before it fetches", async () => {
    fetchMock.mockResolvedValue(jsonResponse([]));
    render(<SearchBox />);

    type("m");
    expect(fetchMock).not.toHaveBeenCalled();

    await vi.advanceTimersByTimeAsync(299);
    expect(fetchMock).not.toHaveBeenCalled();

    // The effect also runs once for the empty query on mount, so the
    // pause ends with two calls, the newest query last (ticket #4821).
    await vi.advanceTimersByTimeAsync(1);
    expect(fetchMock).toHaveBeenCalledTimes(2);
    expect(fetchMock).toHaveBeenLastCalledWith("/api/search?q=m");
  });

  it("renders the titles the API returns", async () => {
    fetchMock.mockResolvedValue(
      jsonResponse([{ title: "Motor mount M8" }, { title: "Motor brush pair" }]),
    );
    render(<SearchBox />);

    type("m");
    await act(async () => {
      await vi.advanceTimersByTimeAsync(300);
    });

    expect(screen.getByText("Motor mount M8")).toBeTruthy();
    expect(screen.getByText("Motor brush pair")).toBeTruthy();
  });

  it("fetches for the empty query on mount", async () => {
    // Documents today's behaviour, not the desired one.
    fetchMock.mockResolvedValue(jsonResponse([]));
    render(<SearchBox />);

    await vi.advanceTimersByTimeAsync(300);
    expect(fetchMock).toHaveBeenCalledTimes(1);
    expect(fetchMock).toHaveBeenCalledWith("/api/search?q=");
  });

  it("fetches again for every keystroke once typing outlasts the debounce", async () => {
    // Also documents today's behaviour: fast or slow, every keystroke
    // costs a request, and a burst of typing is paid back in a late
    // run of them.
    fetchMock.mockResolvedValue(jsonResponse([]));
    render(<SearchBox />);

    type("a");
    await vi.advanceTimersByTimeAsync(350);
    type("ab");
    await vi.advanceTimersByTimeAsync(350);
    type("abc");
    await vi.advanceTimersByTimeAsync(350);

    expect(fetchMock).toHaveBeenCalledTimes(4);
    expect(fetchMock).toHaveBeenNthCalledWith(1, "/api/search?q=");
    expect(fetchMock).toHaveBeenNthCalledWith(2, "/api/search?q=a");
    expect(fetchMock).toHaveBeenNthCalledWith(3, "/api/search?q=ab");
    expect(fetchMock).toHaveBeenNthCalledWith(4, "/api/search?q=abc");
  });
});
~~~
