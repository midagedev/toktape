Then fix it three ways: returning an owned String, working with an index instead of a reference, and splitting it into two functions with a different signature. Say which you would choose and why, make every version handle an empty vector without panicking, and write unit tests for all three.

~~~rust
//! Word-level helpers: normalising tokens and finding the longest word.

/// Lowercases a word and strips the punctuation people type around it,
/// keeping the dots and dashes that belong to part numbers.
///
/// ```rust
/// use wordkit::words::normalize;
///
/// assert_eq!(normalize("\"Bearing!\""), "bearing");
/// assert_eq!(normalize("M8x1.25,"), "m8x1.25");
/// ```
pub fn normalize(word: &str) -> String {
    word.trim()
        .trim_matches(|c: char| !c.is_alphanumeric() && c != '.' && c != '-')
        .to_lowercase()
}

/// Splits text into normalised words, keeping the order they appear in.
/// The notes are plain ASCII prose; no Unicode segmentation yet.
///
/// ```rust
/// use wordkit::words::tokenize;
///
/// assert_eq!(tokenize("Fixed the pump!"), vec!["fixed", "the", "pump"]);
/// ```
pub fn tokenize(text: &str) -> Vec<String> {
    text.split_whitespace().map(normalize).collect()
}

/// Appends an upper-cased copy of the longest word and returns the longest word.
pub fn longest_word(words: &mut Vec<String>) -> &str {
    let mut best = &words[0];
    for w in words.iter() {
        if w.len() > best.len() {
            best = w;
        }
    }
    words.push(best.to_uppercase());
    best
}

src/stats.rs, the counting side:

~~~rust
//! Counting side: how many words, how many distinct, and which ones lead.

use crate::words::tokenize;
use std::collections::BTreeMap;

/// One pass over a text, reduced to the numbers the checker prints.
#[derive(Debug, PartialEq, Eq)]
pub struct WordStats {
    pub words: usize,
    pub distinct: usize,
    pub letters: usize,
}

impl WordStats {
    /// Mean letters per word, `None` for an empty text.
    ///
    /// ```rust
    /// use wordkit::stats;
    ///
    /// let s = stats("abcd ab");
    /// assert_eq!(s.mean_letters(), Some(3.0));
    /// assert_eq!(stats("").mean_letters(), None);
    /// ```
    pub fn mean_letters(&self) -> Option<f64> {
        if self.words == 0 {
            return None;
        }
        Some(self.letters as f64 / self.words as f64)
    }
}

/// Counts words, distinct words and letters in one pass.
///
/// ```rust
/// use wordkit::stats;
///
/// let s = stats("Bearings bearings fail");
/// assert_eq!(s.words, 3);
/// assert_eq!(s.distinct, 2);
/// ```
pub fn stats(text: &str) -> WordStats {
    let words = tokenize(text);
    let mut counts: BTreeMap<&str, ()> = BTreeMap::new();
    let mut letters = 0;
    for word in &words {
        counts.insert(word.as_str(), ());
        letters += word.chars().filter(|c| c.is_alphabetic()).count();
    }
    WordStats {
        words: words.len(),
        distinct: counts.len(),
        letters,
    }
}

/// The `n` most frequent words, ties broken by first appearance.
///
/// ```rust
/// use wordkit::top_words;
///
/// let top = top_words("b a b a c", 2);
/// assert_eq!(top, vec![("b".to_string(), 2), ("a".to_string(), 2)]);
/// ```
pub fn top_words(text: &str, n: usize) -> Vec<(String, usize)> {
    let words = tokenize(text);
    let mut counts: BTreeMap<&str, (usize, usize)> = BTreeMap::new();
    for (i, word) in words.iter().enumerate() {
        counts.entry(word.as_str()).or_insert((0, i)).0 += 1;
    }
    let mut ranked: Vec<(&str, usize, usize)> =
        counts.iter().map(|(w, (c, i))| (*w, *c, *i)).collect();
    ranked.sort_by(|a, b| b.1.cmp(&a.1).then_with(|| a.2.cmp(&b.2)));
    ranked.truncate(n);
    ranked
        .into_iter()
        .map(|(w, c, _)| (w.to_string(), c))
        .collect()
}

/// One line per entry — the word, its count, and a bar of `#` scaled to
/// the busiest entry — the shape `wordkit top` prints.
///
/// ```rust
/// use wordkit::histogram;
///
/// let bars = histogram(&[("aa".to_string(), 8), ("b".to_string(), 2)]);
/// assert_eq!(bars.lines().count(), 2);
/// ```
pub fn histogram(entries: &[(String, usize)]) -> String {
    let max = entries.iter().map(|(_, n)| *n).max().unwrap_or(1);
    entries
        .iter()
        .map(|(word, n)| {
            let bars = if max == 0 { 0 } else { (n * 24).div_ceil(max) };
            format!("{word:<14}{n:>4}  {}", "#".repeat(bars))
        })
        .collect::<Vec<_>>()
        .join("\n")
}

src/shout.rs, new on the branch — the caller the reworked signature has to satisfy:

~~~rust
//! The shout pass the release-notes checker runs after counting: grow
//! the working set with an upper-cased copy of the longest word, then
//! build the one-line report from what came back. Everything here is
//! shaped around the borrow `words.rs` is being refused. The `&str`
//! that `longest_word` hands back carries the lifetime of the `&mut
//! Vec<String>` it was called with, so for as long as the caller holds
//! the reference, the set behind it is borrowed mutably and cannot be
//! read, indexed, or borrowed again.

use crate::words::longest_word;

/// The report line for one pass: the longest word, its length in
/// letters, and how many entries the working set held before the
/// upper-cased copy was appended.
///
/// Everything still needed from `words` is read before the call,
/// because nothing may touch the set again while the returned
/// reference is alive.
///
/// ```rust
/// use wordkit::shout::report_line;
///
/// let mut words = vec!["pump".to_string(), "bearings".to_string()];
/// let line = report_line(&mut words);
/// assert_eq!(line, "bearings (8 letters, shouted from 2)");
/// assert_eq!(words, vec!["pump", "bearings", "BEARINGS"]);
/// ```
pub fn report_line(words: &mut Vec<String>) -> String {
    let before = words.len();
    let longest = longest_word(words);
    format!("{longest} ({} letters, shouted from {before})", longest.chars().count())
}

/// The index of the longest entry, the first of equals winning — the
/// same choice `longest_word` makes. An index is the one fact about
/// the longest word that does not borrow the set to state it, so it
/// stays valid after the set has been grown again.
///
/// ```rust
/// use wordkit::shout::longest_at;
///
/// let words = vec!["pump".to_string(), "bearings".to_string()];
/// assert_eq!(longest_at(&words), Some(1));
/// assert_eq!(longest_at(&[]), None);
/// ```
pub fn longest_at(words: &[String]) -> Option<usize> {
    let mut best: Option<usize> = None;
    for (i, w) in words.iter().enumerate() {
        if best.is_none_or(|b| w.len() > words[b].len()) {
            best = Some(i);
        }
    }
    best
}

src/main.rs, the CLI the subcommand was added to:

~~~rust
use std::env;
use std::io::Read;

use wordkit::{histogram, longest_word, stats, tokenize, top_words};

const USAGE: &str = "\
wordkit 0.2.0 - word counts for release notes

usage:
  wordkit stats [FILE]           words, distinct words, letters, mean length
  wordkit top N [FILE]           the N most frequent words with a bar each
  wordkit shout-longest [FILE]   append an upper-cased copy of the longest
                                 word to the word list, print the longest

reads stdin when FILE is missing or '-'.\
";

fn read_input(path: Option<&str>) -> Result<String, String> {
    match path {
        None | Some("-") => {
            let mut buf = String::new();
            std::io::stdin()
                .read_to_string(&mut buf)
                .map_err(|e| format!("reading stdin: {e}"))?;
            Ok(buf)
        }
        Some(p) => std::fs::read_to_string(p).map_err(|e| format!("reading {p}: {e}")),
    }
}

fn main() {
    std::process::exit(run(&env::args().skip(1).collect::<Vec<_>>()));
}

fn run(args: &[String]) -> i32 {
    let Some(command) = args.first() else {
        eprint!("{USAGE}");
        return 2;
    };
    match command.as_str() {
        "stats" => match read_input(args.get(1).map(String::as_str)) {
            Ok(text) => {
                let s = stats(&text);
                println!("words {}", s.words);
                println!("distinct {}", s.distinct);
                println!("letters {}", s.letters);
                match s.mean_letters() {
                    Some(m) => println!("mean letters {m:.2}"),
                    None => println!("mean letters -"),
                }
                0
            }
            Err(e) => {
                eprintln!("wordkit: {e}");
                1
            }
        },
        "top" => {
            let Some(n) = args.get(1).and_then(|a| a.parse::<usize>().ok()) else {
                eprintln!("wordkit: top needs a number, got {:?}", args.get(1));
                return 2;
            };
            match read_input(args.get(2).map(String::as_str)) {
                Ok(text) => {
                    print!("{}", histogram(&top_words(&text, n)));
                    println!();
                    0
                }
                Err(e) => {
                    eprintln!("wordkit: {e}");
                    1
                }
            }
        }
        "shout-longest" => match read_input(args.get(1).map(String::as_str)) {
            Ok(text) => {
                let mut words = tokenize(&text);
                if words.is_empty() {
                    eprintln!("wordkit: no words in the input");
                    return 1;
                }
                println!("{}", longest_word(&mut words));
                0
            }
            Err(e) => {
                eprintln!("wordkit: {e}");
                1
            }
        },
        "-h" | "--help" => {
            print!("{USAGE}");
            0
        }
        other => {
            eprintln!("wordkit: unknown command {other:?}");
            eprint!("{USAGE}");
            2
        }
    }
}

use std::io::Write;
use std::process::{Command, Stdio};

fn run(args: &[&str], input: &str) -> (String, String, Option<i32>) {
    let mut child = Command::new(env!("CARGO_BIN_EXE_wordkit"))
        .args(args)
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .stderr(Stdio::piped())
        .spawn()
        .expect("spawn wordkit");
    child
        .stdin
        .as_mut()
        .expect("stdin")
        .write_all(input.as_bytes())

    let out = child.wait_with_output().expect("wait");
    (
        String::from_utf8_lossy(&out.stdout).into_owned(),
        String::from_utf8_lossy(&out.stderr).into_owned(),
        out.status.code(),
    )
}
