The whole script, as it runs today:

~~~python
#!/usr/bin/env python3
"""Summarise an nginx timing log: request count and p95 latency per route.

Reads one log file, groups request times by route path, and prints one
line per route: the path, the number of requests, and the 95th
percentile latency in milliseconds.
"""

import re
import sys
from collections import defaultdict

# One line of the archived format the shipper writes (see the nginx
# log_format and the vector remap below): remote addr, ident, user,
# [time], "METHOD URI PROTO", status, bytes, elapsed-milliseconds+ms.
LINE_RE = re.compile(
    r'(\S+) \S+ \S+ \[(.*?)\] "(\w+) (\S+) \S+" (\d+) (\d+) (\d+)ms'
)


def p95(values):
    values.sort()
    return values[int(len(values) * 0.95)]


def route_of(uri):
    """The grouping key: the URI without its query string."""
    return uri.split("?")[0]


def parse_line(line):
    """Return (route, elapsed_ms) for a matching line, else None."""
    m = LINE_RE.match(line)
    if not m:
        return None
    return route_of(m.group(4)), int(m.group(7))


def main(path):
    lines = open(path).readlines()
    by_route = {}
    for line in lines:
        hit = parse_line(line)
        if hit is None:
            continue
        route, elapsed = hit
        by_route.setdefault(route, []).append(elapsed)
    for route, times in by_route.items():
        print(route, len(times), p95(times))


if __name__ == "__main__":
    if len(sys.argv) != 2:
        sys.exit(f"usage: {sys.argv[0]} <access-log>")
    main(sys.argv[1])
~~~

The test file that passes against it today:

~~~python
import route_summary as rs


def test_parse_line_extracts_route_and_milliseconds():
    line = ('10.2.19.144 - - [18/Sep/2026:12:03:41 +0000] '
            '"POST /api/cart/items HTTP/1.1" 201 512 43ms\n')
    assert rs.parse_line(line) == ("/api/cart/items", 43)


def test_parse_line_strips_the_query_string():
    line = ('10.2.19.151 - - [18/Sep/2026:12:03:41 +0000] '
            '"GET /api/cart/items?limit=50 HTTP/1.1" 200 4310 18ms\n')
    assert rs.parse_line(line) == ("/api/cart/items", 18)


def test_parse_line_rejects_handshake_noise():
    line = ('10.2.19.99 - - [18/Sep/2026:12:03:41 +0000] '
            '"\\x16\\x03\\x01\\x8f" 400 157 0ms\n')
    assert rs.parse_line(line) is None


def test_p95_of_three_values_returns_the_max():
    assert rs.p95([30, 50, 40]) == 50


def test_main_prints_one_line_per_route(tmp_path, capsys):
    log = tmp_path / "mini.log"
    log.write_text(
        '10.0.0.1 - - [18/Sep/2026:12:03:41 +0000] "GET /health HTTP/1.1" 200 2 3ms\n'
        '10.0.0.2 - - [18/Sep/2026:12:03:41 +0000] "GET /api/cart HTTP/1.1" 200 90 12ms\n'
        '10.0.0.3 - - [18/Sep/2026:12:03:41 +0000] "GET /api/cart HTTP/1.1" 200 91 34ms\n'
    )
    rs.main(str(log))
    out = capsys.readouterr().out.splitlines()
    assert out[0] == "/health 1 3"
    assert out[1] == "/api/cart 2 34"
~~~

How the log gets its shape. The load balancers write the timing format; the shipper merges all four of them, converts nginx's fractional seconds to whole milliseconds, and writes one archive file per day:

~~~nginx
# /etc/nginx/nginx.conf, logging half
http {
    log_format timing '$remote_addr - $remote_user [$time_local] '
                      '"$request" $status $body_bytes_sent $request_time';

    access_log /var/log/nginx/timing.log timing;
}
~~~

~~~toml
# /etc/vector/vector.toml — nginx timing log to the archive format
[sources.nginx_timing]
type        = "file"
include     = ["/var/log/nginx/timing.log"]
read_from   = "beginning"
ignore_older = 86400

[transforms.archive_format]
type   = "remap"
inputs = ["nginx_timing"]
source = '''
m = parse_regex!(.message, r'^(?P<addr>\S+) (?P<ident>\S+) (?P<user>\S+) \[(?P<time>[^\]]+)\] "(?P<request>[^"]*)" (?P<status>\d+) (?P<bytes>\d+) (?P<elapsed_s>\d+\.\d+)$')
elapsed_ms = to_int!(round(to_float!(m.elapsed_s) * 1000.0))
.line = join!([m.addr, m.ident, m.user, "[" + m.time + "]", '"' + m.request + '"', m.status, m.bytes, to_string!(elapsed_ms) + "ms"], " ")
'''

[sinks.archive]
type          = "file"
inputs        = ["archive_format"]
path          = "/mnt/archive/access-%Y%m%d.log"
encoding.codec = "text"
~~~
