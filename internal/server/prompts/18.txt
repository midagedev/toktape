This Django view renders the order history page. It takes four seconds for customers with many orders, and the database log shows 1,300 queries for a page load. Explain where each query comes from and how the count grows with the orders and items. Point out the bug in the shipment line when Shipment has a OneToOneField to Order. The view itself, orders/views.py, and the URLconf that routes it:

~~~python
from django.shortcuts import render

from .models import Order


def order_history(request):
    orders = Order.objects.filter(customer=request.user.customer).order_by("-created_at")
    rows = []
    for order in orders:
        items = order.items.all()
        rows.append({
            "id": order.id,
            "date": order.created_at,
            "total": sum(i.price * i.quantity for i in items),
            "status": order.shipment.status if order.shipment else "pending",
            "products": [i.product.name for i in items],
        })
    return render(request, "orders/history.html", {"rows": rows})
~~~

~~~python
# shipping/migrations/0002_backfill_orders.py
from django.db import migrations


def backfill(apps, schema_editor):
    """Every order that predates the shipping app gets a shipment row,
    so the history page's status column is never empty for old orders."""
    Order = apps.get_model("orders", "Order")
    Shipment = apps.get_model("shipping", "Shipment")
    for order in Order.objects.filter(shipment__isnull=True).iterator():
        Shipment.objects.create(
            order=order,
            carrier="legacy",
            status="delivered",
            shipped_at=order.created_at,
        )


class Migration(migrations.Migration):

    dependencies = [("shipping", "0001_initial")]

    operations = [
        migrations.RunPython(backfill, migrations.RunPython.noop),
    ]
~~~

And the test file that passes today, orders/tests.py — every order it creates gets a shipment, because the fixture shape was copied from production after the backfill:

~~~python
from decimal import Decimal

from django.contrib.auth.models import User
from django.test import TestCase

from catalog.models import Product
from shipping.models import Shipment

from .models import Customer, Order, OrderItem


def make_order(customer, products, status="in_transit"):
    order = Order.objects.create(customer=customer)
    for product, price, qty in products:
        OrderItem.objects.create(
            order=order, product=product, price=price, quantity=qty
        )
    Shipment.objects.create(order=order, carrier="parcelco", status=status)
    return order


class OrderHistoryViewTests(TestCase):
    def setUp(self):
        self.user = User.objects.create_user("dana")
        self.customer = Customer.objects.create(user=self.user)
        self.grip = Product.objects.create(sku="GRIP-2", name="Bar tape")
        self.tube = Product.objects.create(sku="TUBE-7", name="Inner tube")

    def test_page_lists_orders_with_totals_and_statuses(self):
        make_order(
            self.customer,
            [(self.grip, Decimal("12.50"), 2), (self.tube, Decimal("4.25"), 1)],
        )
        self.client.force_login(self.user)
        response = self.client.get("/orders/history/")
        self.assertEqual(response.status_code, 200)
        self.assertContains(response, "Order history")
        self.assertContains(response, "29.25")
        self.assertContains(response, "in_transit")
        self.assertContains(response, "Bar tape")
        self.assertContains(response, "Inner tube")

    def test_status_column_reads_each_shipment(self):
        make_order(self.customer, [(self.grip, Decimal("12.50"), 1)], status="delivered")
        make_order(self.customer, [(self.tube, Decimal("4.25"), 2)], status="packed")
        self.client.force_login(self.user)
        response = self.client.get("/orders/history/")
        self.assertContains(response, "delivered")
        self.assertContains(response, "packed")

    def test_page_for_a_customer_with_no_orders(self):
        self.client.force_login(self.user)
        response = self.client.get("/orders/history/")
        self.assertEqual(response.status_code, 200)
        self.assertContains(response, "Order history")
~~~
